http Module

Make HTTP requests to web APIs and servers.

Usage
import http

Functions

FunctionReturnsDescription
http.get(url) dictionary Perform an HTTP GET request
http.get(url, headers) dictionary GET with custom headers dictionary
http.post(url, body) dictionary POST with a string body
http.post(url, body, headers) dictionary POST with body and custom headers
http.put(url, body) dictionary HTTP PUT request
http.delete(url) dictionary HTTP DELETE request

Response dictionary

All functions return a dictionary with these keys:

KeyTypeDescription
"status"numberHTTP status code (e.g. 200, 404)
"body"stringResponse body as a string
"headers"dictionaryResponse headers

GET request

 get.ks
import http
import json

let res = http.get("https://jsonplaceholder.typicode.com/todos/1")

if res["status"] == 200 {
    let data = json.parse(res["body"])
    print(data["title"])
} else {
    print("Request failed: {$res["status"]}")
}

POST request

 post.ks
import http
import json

let payload = json.stringify({
    "title": "My new post",
    "body": "Hello from KelpyShark",
    "userId": 1
})

let headers = {"Content-Type": "application/json"}
let res = http.post("https://jsonplaceholder.typicode.com/posts", payload, headers)

if res["status"] == 201 {
    let created = json.parse(res["body"])
    print("Created post id: {$created["id"]}")
}

Authenticated request

 auth.ks
import http
import json
import sys

let token = sys.env["API_TOKEN"]
let headers = {"Authorization": "Bearer {$token}"}

let res = http.get("https://api.example.com/me", headers)
let me  = json.parse(res["body"])
print("Logged in as: {$me["username"]}")
Note — HTTP functions are synchronous (blocking). They wait for the response before the next line of code runs.
