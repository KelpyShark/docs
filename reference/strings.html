strings Module

String manipulation as free functions.

Usage
import strings

The strings module mirrors the string dot-methods but as standalone functions. Useful when passing string operations as callbacks or building pipelines.

FunctionReturnsDescription
strings.upper(s)stringUppercase version of s
strings.lower(s)stringLowercase version of s
strings.trim(s)stringRemove leading/trailing whitespace
strings.trim_start(s)stringRemove leading whitespace only
strings.trim_end(s)stringRemove trailing whitespace only
strings.replace(s, old, new)stringReplace all occurrences of old with new
strings.split(s, sep)listSplit s on separator sep
strings.join(list, sep)stringJoin list items with sep
strings.contains(s, sub)booleanTrue if sub appears in s
strings.starts_with(s, prefix)booleanTrue if s starts with prefix
strings.ends_with(s, suffix)booleanTrue if s ends with suffix
strings.index_of(s, sub)numberIndex of first sub, or -1
strings.slice(s, start, end)stringSubstring from start to end
strings.repeat(s, n)stringRepeat s n times
strings.reverse(s)stringReverse the characters of s
strings.pad_left(s, n, ch)stringPad s to length n with ch on the left
strings.pad_right(s, n, ch)stringPad s to length n with ch on the right

Examples

 strings-demo.ks
import strings

let words = ["the", "quick", "brown", "fox"]

# Capitalise each word
let capitalised = []
for w in words {
    let first = strings.upper(strings.slice(w, 0, 1))
    let rest  = strings.slice(w, 1, len(w))
    push(capitalised, first + rest)
}
print(strings.join(capitalised, " "))
# The Quick Brown Fox

# CSV parsing
let csv = "Alice,30,Engineer"
let parts = strings.split(csv, ",")
print(parts[0])    # Alice
print(parts[1])    # 30

# Table formatting with padding
let headers = ["Name", "Score"]
for h in headers {
    print(strings.pad_right(h, 10, " ") + "|")
}
