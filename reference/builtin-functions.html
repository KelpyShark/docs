Built-in Functions

Always-available functions — no import needed.

print(value)

Outputs value to the console followed by a newline.

Example
print("Hello")
print(42)
print(true)
Output
Hello
42
true

len(value)

Returns the length of a string or list.

Example
print(len("hello"))       # 5
print(len([1, 2, 3]))     # 3

type(value)

Returns a string describing the type of value.

Example
print(type(42))           # number
print(type("hi"))         # string
print(type(true))         # boolean
print(type(null))         # null
print(type([]))           # list
print(type({}))           # dictionary

str(value)

Converts value to its string representation.

Example
let n = 3.14
let s = str(n)
print(s + " is pi")       # 3.14 is pi

num(value)

Converts a string to a number. Throws an error if the string is not a valid number.

Example
let x = num("42")
print(x + 1)              # 43

let y = num("3.14")
print(y * 2)              # 6.28

bool(value)

Converts a value to boolean. The following are falsy: false, null, 0, "", [].

Example
print(bool(0))            # false
print(bool(1))            # true
print(bool(""))           # false
print(bool("hi"))         # true
print(bool([]))           # false

range(stop) / range(start, stop) / range(start, stop, step)

Returns a list of numbers. Useful with for loops.

Example
print(range(5))           # [0, 1, 2, 3, 4]
print(range(2, 6))        # [2, 3, 4, 5]
print(range(0, 10, 2))    # [0, 2, 4, 6, 8]

for i in range(3) {
    print(i)
}

push(list, value)

Appends value to the end of list. Modifies the list in place.

Example
let nums = [1, 2]
push(nums, 3)
print(nums)               # [1, 2, 3]

pop(list)

Removes and returns the last element of list.

Example
let items = ["a", "b", "c"]
let last = pop(items)
print(last)               # c
print(items)              # ["a", "b"]
