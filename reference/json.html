json Module

Parse and serialise JSON data.

Usage
import json
FunctionReturnsDescription
json.parse(str) any Parse a JSON string into KelpyShark values. Throws on invalid JSON.
json.stringify(value) string Convert a KelpyShark value to a JSON string.
json.stringify_pretty(value) string Like stringify but with indentation for readability.

Type mapping

JSONKelpyShark
object {}dictionary
array []list
stringstring

Parsing JSON

 parse.ks
import json

let raw = '{"name":"Alice","scores":[95,88,72],"active":true}'
let data = json.parse(raw)

print(data["name"])         # Alice
print(data["scores"][0])    # 95
print(data["active"])       # true

Serialising to JSON

 stringify.ks
import json

let user = {
    "name": "Bob",
    "age": 25,
    "hobbies": ["chess", "cycling"]
}

let compact = json.stringify(user)
print(compact)
# {"name":"Bob","age":25,"hobbies":["chess","cycling"]}

let pretty = json.stringify_pretty(user)
print(pretty)
# {
#   "name": "Bob",
#   "age": 25,
#   "hobbies": [
#     "chess",
#     "cycling"
#   ]
# }

Read / modify / write a JSON file

 config.ks
import io
import json

# Load config
let raw  = io.read("config.json")
let cfg  = json.parse(raw)

# Modify
cfg["version"] = "2.0"

# Save back
io.write("config.json", json.stringify_pretty(cfg))
