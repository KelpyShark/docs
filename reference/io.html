io Module

File reading, writing, and filesystem queries.

Usage
import io
FunctionReturnsDescription
io.read(path)stringRead entire file contents as a string. Throws if file not found.
io.write(path, content)nullWrite (overwrite) a file with content.
io.append(path, content)nullAppend content to a file (creates if not present).
io.exists(path)booleanTrue if the file or directory exists.
io.delete(path)nullDelete the file at path. Throws if not found.
io.input(prompt)stringDisplay prompt and return a line of user input.
io.lines(path)listRead a file and return each line as a list element.

Examples

 io-demo.ks
import io

# Write a file
io.write("hello.txt", "Hello, KelpyShark!\n")

# Append to it
io.append("hello.txt", "Second line\n")

# Read it back
let content = io.read("hello.txt")
print(content)

# Read line by line
let lines = io.lines("hello.txt")
for line in lines {
    print(">> {$line}")
}

# File existence check
if io.exists("hello.txt") {
    print("File found!")
}

# User input
let name = io.input("Enter your name: ")
print("Welcome, {$name}!")
Warning — io.write() overwrites the file completely. Use io.append() if you want to add content without losing existing data.

Reading a CSV file

 csv.ks
import io
import strings

let rows = io.lines("data.csv")
for row in rows {
    let cols = strings.split(row, ",")
    print("Name: {$cols[0]}  Score: {$cols[1]}")
}

Error handling with io

import io

try {
    let data = io.read("missing.txt")
    print(data)
} catch (e) {
    print("File error: {$e}")
}
