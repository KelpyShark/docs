Examples

 math-demo.ks
import math

# Circle area
def circle_area(r) {
    return math.pi * math.pow(r, 2)
}
print(circle_area(5))          # 78.53981633974483

# Hypotenuse
def hyp(a, b) {
    return math.sqrt(math.pow(a, 2) + math.pow(b, 2))
}
print(hyp(3, 4))               # 5

# Clamping a value
def clamp(val, lo, hi) {
    return math.max(lo, math.min(hi, val))
}
print(clamp(15, 0, 10))        # 10
print(clamp(-3, 0, 10))        # 0

# Random dice roll
let roll = math.random_int(1, 6)
print("Rolled: {$roll}")
